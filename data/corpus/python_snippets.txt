def is_even(n):
    return n % 2 == 0

def is_odd(n):
    return n % 2 != 0

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def is_perfect_square(n):
    if n < 0:
        return False
    root = int(n ** 0.5)
    return root * root == n or (root + 1) ** 2 == n

def is_perfect_number(n):
    if n < 2:
        return False
    total = sum(d for d in range(1, n) if n % d == 0)
    return total == n

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    x, y = a, b
    while y:
        x, y = y, x % y
    return abs(a * b) // x

def is_leap_year(year):
    return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)

def digit_sum(n):
    return sum(int(d) for d in str(abs(n)))

def digit_count(n):
    return len(str(abs(n)))

def reverse_int(n):
    sign = -1 if n < 0 else 1
    return sign * int(str(abs(n))[::-1])

def is_palindrome_number(n):
    s = str(n)
    return s == s[::-1]

def sum_of_squares(n):
    return sum(i * i for i in range(1, n + 1))

def factorial(n):
    if n == 0:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

def nth_triangular_number(n):
    return n * (n + 1) // 2

def is_armstrong_number(n):
    digits = str(n)
    power = len(digits)
    return sum(int(d) ** power for d in digits) == n

def clamp(x, lo, hi):
    return max(lo, min(x, hi))

def average(nums):
    return sum(nums) / len(nums)

def is_multiple(n, k):
    return n % k == 0

def power(base, exp):
    result = 1
    for _ in range(exp):
        result *= base
    return result

def sign(n):
    if n > 0:
        return 1
    elif n < 0:
        return -1
    return 0

def absolute_value(n):
    return n if n >= 0 else -n

def celsius_to_fahrenheit(c):
    return c * 9 / 5 + 32

def fahrenheit_to_celsius(f):
    return (f - 32) * 5 / 9

def is_power_of_two(n):
    return n > 0 and (n & (n - 1)) == 0

def reverse_string(s):
    return s[::-1]

def is_palindrome(s):
    return s == s[::-1]

def count_vowels(s):
    return sum(1 for c in s.lower() if c in "aeiou")

def count_consonants(s):
    return sum(1 for c in s.lower() if c.isalpha() and c not in "aeiou")

def is_anagram(a, b):
    return sorted(a) == sorted(b)

def word_count(s):
    return len(s.split())

def capitalize_words(s):
    return " ".join(word.capitalize() for word in s.split())

def is_upper(s):
    return s == s.upper()

def is_lower(s):
    return s == s.lower()

def remove_whitespace(s):
    return "".join(s.split())

def count_occurrences_char(s, ch):
    return s.count(ch)

def longest_word(s):
    words = s.split()
    return max(words, key=len)

def is_numeric_string(s):
    return s.isdigit()

def char_frequency(s):
    freq = {}
    for c in s:
        freq[c] = freq.get(c, 0) + 1
    return freq

def is_pangram(s):
    return set("abcdefghijklmnopqrstuvwxyz") <= set(s.lower())

def caesar_cipher(s, shift):
    result = ""
    for c in s:
        if c.isalpha():
            base = ord("A") if c.isupper() else ord("a")
            result += chr((ord(c) - base + shift) % 26 + base)
        else:
            result += c
    return result

def rot13(s):
    result = ""
    for c in s:
        if c.isalpha():
            base = ord("A") if c.isupper() else ord("a")
            result += chr((ord(c) - base + 13) % 26 + base)
        else:
            result += c
    return result

def count_uppercase(s):
    return sum(1 for c in s if c.isupper())

def first_non_repeating_char(s):
    for c in s:
        if s.count(c) == 1:
            return c
    return None

def swap_case(s):
    return s.swapcase()

def starts_with_vowel(s):
    return bool(s) and s[0].lower() in "aeiou"

def sum_list(nums):
    total = 0
    for x in nums:
        total += x
    return total

def max_of_list(nums):
    best = nums[0]
    for x in nums[1:]:
        if x > best:
            best = x
    return best

def min_of_list(nums):
    best = nums[0]
    for x in nums[1:]:
        if x < best:
            best = x
    return best

def unique_chars(s):
    return len(set(s)) == len(s)

def flatten(nested):
    result = []
    for item in nested:
        if isinstance(item, list):
            result.extend(flatten(item))
        else:
            result.append(item)
    return result

def chunk_list(lst, size):
    return [lst[i:i + size] for i in range(0, len(lst), size)]

def rotate_list(lst, k):
    if not lst:
        return lst
    k = k % len(lst)
    return lst[k:] + lst[:k]

def is_sorted(lst):
    return all(lst[i] <= lst[i + 1] for i in range(len(lst) - 1))

def second_largest(nums):
    uniq = sorted(set(nums), reverse=True)
    return uniq[1]

def most_frequent(lst):
    return max(set(lst), key=lst.count)

def remove_duplicates_list(lst):
    seen = []
    for x in lst:
        if x not in seen:
            seen.append(x)
    return seen

def count_occurrences_list(lst, item):
    return lst.count(item)

def merge_sorted(a, b):
    result = []
    i = j = 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            result.append(a[i])
            i += 1
        else:
            result.append(b[j])
            j += 1
    result.extend(a[i:])
    result.extend(b[j:])
    return result

def intersection_of_lists(a, b):
    return sorted(set(a) & set(b))

def union_of_lists(a, b):
    return sorted(set(a) | set(b))

def difference_of_lists(a, b):
    return sorted(set(a) - set(b))

def list_product(nums):
    result = 1
    for x in nums:
        result *= x
    return result

def cumulative_sum(nums):
    result = []
    total = 0
    for x in nums:
        total += x
        result.append(total)
    return result

def find_index(lst, target):
    for i, x in enumerate(lst):
        if x == target:
            return i
    return -1

def contains_duplicate(lst):
    return len(lst) != len(set(lst))

def is_subset(a, b):
    return set(a) <= set(b)

def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

def selection_sort(arr):
    n = len(arr)
    for i in range(n):
        min_idx = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_idx]:
                min_idx = j
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
    return arr

def insertion_sort(arr):
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr

def binary_search(arr, target):
    lo, hi = 0, len(arr) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

def find_min_max(nums):
    return min(nums), max(nums)

class Stack:
    def __init__(self):
        self.items = []

    def push(self, item):
        self.items.append(item)

    def pop(self):
        return self.items.pop()

    def is_empty(self):
        return len(self.items) == 0

class Queue:
    def __init__(self):
        self.items = []

    def enqueue(self, item):
        self.items.append(item)

    def dequeue(self):
        return self.items.pop(0)

    def is_empty(self):
        return len(self.items) == 0
